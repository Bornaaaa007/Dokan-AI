'use client';
import { useState } from 'react';
import Link from 'next/link';
import { useRouter } from 'next/navigation';
import { supabase } from '../../lib/supabaseClient';
import { api } from '../../lib/api';

const emailOk = (v) => /^\S+@\S+\.\S+$/.test(v);

export default function Login() {
  const router = useRouter();
  const [mode, setMode] = useState('login'); // 'login' | 'signup'
  const [role, setRole] = useState('customer');
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [show, setShow] = useState(false);
  const [terms, setTerms] = useState(false);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState('');
  const [info, setInfo] = useState('');

  const isSignup = mode === 'signup';
  const valid = emailOk(email) && password.length >= 6 && (!isSignup || (name.trim().length > 1 && terms));
  const reset = (m) => { setMode(m); setError(''); setInfo(''); };

 // Ask the backend who this user is, then send them to the right place
const go = async () => {
  try {
    const { data } = await api('/me');
    const roles = data.roles || [];
    router.push(roles.includes('admin') ? '/admin' : roles.includes('seller') ? '/seller' : '/shop');
  } catch {
    router.push('/shop');
  }
  router.refresh();
};

  const submit = async (e) => {
    e.preventDefault();
    if (!valid) return;
    setError(''); setInfo(''); setBusy(true);
    try {
      if (isSignup) {
        const { data, error } = await supabase.auth.signUp({ email, password, options: { data: { name, role } } });
        if (error) throw error;
        if (!data.session) {
          setInfo('Account created. Check your email to confirm, then log in.');
          setMode('login');
          return;
        }
        await go();
      } else {
        const { data, error } = await supabase.auth.signInWithPassword({ email, password });
        if (error) throw error;
        await go();
      }
    } catch (err) {
      setError(err.message || 'Something went wrong. Please try again.');
    } finally {
      setBusy(false);
    }
  };

  const google = async () => {
    setError('');
    const { error } = await supabase.auth.signInWithOAuth({ provider: 'google', options: { redirectTo: `${window.location.origin}/shop` } });
    if (error) setError(error.message);
  };

  const forgot = async () => {
    setError(''); setInfo('');
    if (!emailOk(email)) return setError('Enter your email above first.');
    const { error } = await supabase.auth.resetPasswordForEmail(email, { redirectTo: `${window.location.origin}/login` });
    if (error) setError(error.message); else setInfo('Password reset link sent. Check your email.');
  };

  return (
    <div className="auth">
      <aside className="auth-brand">
        <Link href="/" className="logo">Dokan<b>AI</b></Link>
        <h2>Your shop. Your stylist. Your trust.</h2>
        <ul>
          <li><b>Shop smarter</b>Complete outfits within your budget, from verified sellers.</li>
          <li><b>Sell with confidence</b>Orders, stock and payouts in one place.</li>
          <li><b>Protected payments</b>Funds stay in escrow until delivery is confirmed.</li>
        </ul>
      </aside>

      <main className="auth-main">
        <div className="auth-card">
          <form onSubmit={submit}>
            <div className="seg" role="tablist">
              {[['login', 'Log in'], ['signup', 'Sign up']].map(([k, l]) => (
                <button type="button" key={k} className={mode === k ? 'on' : ''} onClick={() => reset(k)}>{l}</button>
              ))}
            </div>
            <h1>{isSignup ? 'Create your account' : 'Welcome back'}</h1>
            <p className="sub">{isSignup ? 'It takes less than a minute.' : 'Log in to continue.'}</p>

            {isSignup && <>
              <div className="roles">
                {[['customer', '🛍️', 'Customer', 'Shop and get styled'], ['seller', '🏪', 'Seller', 'Open your shop']].map(([k, ic, t, d]) => (
                  <button type="button" key={k} className={role === k ? 'on' : ''} onClick={() => setRole(k)}><span>{ic}</span><b>{t}</b><small>{d}</small></button>
                ))}
              </div>
              <label>Full name<input value={name} onChange={(e) => setName(e.target.value)} placeholder="Your name" autoComplete="name" /></label>
            </>}

            <label>Email address<input type="email" placeholder="you@example.com" value={email} onChange={(e) => setEmail(e.target.value)} autoComplete="email" /></label>
            <label>Password
              <div className="pfx">
                <input type={show ? 'text' : 'password'} placeholder="Min 6 characters" value={password} onChange={(e) => setPassword(e.target.value)} autoComplete={isSignup ? 'new-password' : 'current-password'} style={{ paddingLeft: 12 }} />
                <button type="button" className="link" onClick={() => setShow(!show)} style={{ padding: '0 12px' }}>{show ? 'Hide' : 'Show'}</button>
              </div>
            </label>
            {!isSignup && <button type="button" className="link" style={{ justifySelf: 'end', fontSize: 13 }} onClick={forgot}>Forgot password?</button>}

            {isSignup && <label className="chk"><input type="checkbox" checked={terms} onChange={(e) => setTerms(e.target.checked)} /> <span>I agree to the Terms and Privacy Policy.</span></label>}
            {isSignup && role === 'seller' && <p className="note">After signup you will verify your business (NID, shop page, payout account) before you can sell.</p>}

            {error && <p className="err">{error}</p>}
            {info && <p style={{ color: 'var(--ok)', margin: 0, fontSize: 14 }}>{info}</p>}

            <button className="btn" type="submit" disabled={!valid || busy} style={{ opacity: !valid || busy ? 0.5 : 1 }}>
              {busy ? 'Please wait…' : isSignup ? 'Create account' : 'Log in'}
            </button>

// Pure functions: turn rows from the analytics views into API responses and plain-language tips.
// No database, no network: everything here is unit-tested.

export const isPremium = (shop, now = new Date()) =>
  shop?.plan === 'premium' && (!shop.plan_expires_at || new Date(shop.plan_expires_at) > now);

const plural = (n, word) => n + ' ' + word + (n === 1 ? '' : 's');
const demandName = (r) => [r.occasion, r.color, r.custom_category].filter(Boolean).join(' ');
const byDesc = (key) => (a, b) => (b[key] ?? 0) - (a[key] ?? 0);

function demandTip(r) {
  const name = demandName(r);
  const supply = r.listings === 0 ? 'no shop has it in stock'
    : r.listings === 1 ? 'only 1 in-stock listing exists'
    : 'only ' + r.listings + ' in-stock listings exist';
  return 'Customers searched for "' + name + '" ' + r.searches_30d + ' times in the last 30 days, but ' + supply + '. Stocking it could win these buyers.';
}

export function demandList(demand, limit = 5) {
  return demand
    .filter((r) => r.signal !== 'not_enough_data')
    .sort(byDesc('opportunity_score'))
    .slice(0, limit)
    .map((r) => ({ name: demandName(r), custom_category: r.custom_category, color: r.color, occasion: r.occasion, searches_30d: r.searches_30d, listings: r.listings, signal: r.signal }));
}

// Seller view: categories only, so one shop never sees another shop's products or name.
export function trendingCategories(trending, limit = 5) {
  const map = new Map();
  for (const t of trending.filter((x) => x.signal === 'trending')) {
    const key = t.custom_category ?? 'other';
    const cur = map.get(key) ?? { custom_category: key, trending_products: 0, score_7d: 0 };
    cur.trending_products += 1;
    cur.score_7d += t.score_7d;
    map.set(key, cur);
  }
  return [...map.values()].sort(byDesc('score_7d')).slice(0, limit);
}

export function sellerInsights({ performance, trending, demand }) {
  const out = [];
  for (const r of demand.filter((d) => d.signal === 'high_opportunity').sort(byDesc('opportunity_score')).slice(0, 2))
    out.push({ type: 'demand_gap', title: 'Demand gap', text: demandTip(r), product_id: null });
  for (const p of performance.filter((x) => x.signal === 'doing_badly').sort(byDesc('views_30d')).slice(0, 2))
    out.push({ type: 'underperformer', title: 'Underperformer', text: p.name + ' got ' + plural(p.views_30d, 'view') + ' but no sales in 30 days. Try a lower price or better photos.', product_id: p.product_id });
  for (const p of performance.filter((x) => !x.in_stock && x.purchases_30d > 0).sort(byDesc('purchases_30d')).slice(0, 2))
    out.push({ type: 'restock', title: 'Restock', text: p.name + ' sold ' + plural(p.purchases_30d, 'time') + ' in 30 days but is out of stock. Restock it.', product_id: p.product_id });
  const mine = new Set(performance.map((p) => p.product_id));
  for (const t of trending.filter((x) => x.signal === 'trending' && mine.has(x.product_id)).sort(byDesc('score_7d')).slice(0, 1))
    out.push({ type: 'trending', title: 'Trending', text: t.name + ' is trending this week. Keep it in stock.', product_id: t.product_id });
  return out;
}

export function buildSellerAnalytics({ shop, performance, trending = [], demand = [], topSellers = [], now = new Date() }) {
  const premium = isPremium(shop, now);
  const views = performance.reduce((t, p) => t + p.views_30d, 0);
  const carts = performance.reduce((t, p) => t + p.carts_30d, 0);
  const purchases = performance.reduce((t, p) => t + p.purchases_30d, 0);
  const best = performance.filter((p) => p.purchases_30d > 0).sort((a, b) => b.purchases_30d - a.purchases_30d || b.views_30d - a.views_30d);
  const rank = topSellers.find((t) => t.shop_id === shop.id)?.rnk ?? null;

  const base = {
    shop: { id: shop.id, name: shop.name, plan: premium ? 'premium' : 'free', top_seller_rank: rank },
    summary: {
      views_30d: views, carts_30d: carts, purchases_30d: purchases,
      conversion: views > 0 ? Math.round((purchases / views) * 1000) / 1000 : null,
      best_seller: best[0] ? { product_id: best[0].product_id, name: best[0].name, purchases_30d: best[0].purchases_30d } : null,
    },
    locked: !premium,
  };
  if (!premium) return { ...base, best_products: [], worst_products: [], insights: [], platform: null };

  const slim = (p) => ({ product_id: p.product_id, name: p.name, views_30d: p.views_30d, carts_30d: p.carts_30d, purchases_30d: p.purchases_30d, conversion: p.conversion, in_stock: p.in_stock });
  return {
    ...base,
    best_products: performance.filter((p) => p.signal === 'doing_well').sort(byDesc('purchases_30d')).slice(0, 3).map((p) => ({ ...slim(p), signal: p.signal })),
    worst_products: performance.filter((p) => p.signal === 'doing_badly').sort(byDesc('views_30d')).slice(0, 3).map((p) => ({ ...slim(p), signal: p.signal })),
    no_data_products: performance.filter((p) => p.signal === 'new_no_views' || p.signal === 'not_enough_data').length,
    insights: sellerInsights({ performance, trending, demand }),
    platform: { demand_gap: demandList(demand), trending_categories: trendingCategories(trending) },
  };
}

export class HttpError extends Error {
  constructor(status, code, message) { super(message); this.status = status; this.code = code; }
}

export const resolveDb = (d) => (typeof d === 'function' ? d() : d);

export async function run(dbOrFactory, fn, label = 'api') {

    return await fn(resolveDb(dbOrFactory));

    console.error(label + ' error:', e);   // details stay in the server log, never in the reply
    return json(500, { error: { code: 'internal', message: 'Something went wrong. Please try again.' } });
  }
}

// ONE place that answers "who is calling?". Every new route uses this, so real login only has to change this file.
//
// 1) Authorization: Bearer <token>  -> db.getUserByToken(token). Written for Supabase Auth
//    (users.auth_id links to auth.users). If the auth teammate issues their own tokens, change getUserByToken in db.mjs.
// 2) x-user-id header (demo mode)   -> db.getUser(id).
//    TODO(auth): delete this branch once login works, or set DISABLE_DEV_USER_HEADER=true in production.
import { UUID, http } from '../api/http.mjs';

export async function currentUser(request, db) {
  const auth = request.headers.get('authorization');
  if (auth) {
    const m = /^Bearer\s+(\S+)$/i.exec(auth.trim());
    if (!m || typeof db.getUserByToken !== 'function') return null;   // a bad token never falls back to the demo header
    return (await db.getUserByToken(m[1])) ?? null;
  }
  if (process.env.DISABLE_DEV_USER_HEADER === 'true') return null;
  const id = request.headers.get('x-user-id');
  if (!id || !UUID.test(id)) return null;
  return (await db.getUser(id)) ?? null;
}

export async function requireUser(request, db) {
  const user = await currentUser(request, db);
  if (!user) throw http(401, 'not_signed_in', 'Please sign in first.');
  return user;
}

// Complaint rules in one place. Pure functions, no database.

export const REASONS = {
  customer: ['not_delivered', 'not_as_described', 'wrong_size', 'other'],
  seller: ['refused_delivery', 'false_claim', 'other'],
};
export const LIMITS = {
  textMin: 10, textMax: 2000,
  evidenceMax: 10, urlMax: 500,
  filesPer24h: 5,
  responseHours: 48,
};
export const BADGE_LABEL = 'Verified complaints';
export const BADGE_LEVEL = 'fraud';

const ok = (value) => ({ ok: true, value });
const bad = (message) => ({ ok: false, message });

export const responseDueAt = (createdAt) => new Date(new Date(createdAt).getTime() + LIMITS.responseHours * 3600 * 1000);

function cleanText(v, label) {
  if (typeof v !== 'string') return bad(label + ' is required.');
  const t = v.trim();
  if (t.length < LIMITS.textMin) return bad(label + ' must be at least ' + LIMITS.textMin + ' characters.');
  if (t.length > LIMITS.textMax) return bad(label + ' must be at most ' + LIMITS.textMax + ' characters.');
  return ok(t);
}

export function cleanEvidence(v) {
  if (v === undefined || v === null) return ok([]);
  if (!Array.isArray(v)) return bad('evidence_urls must be a list of links.');
  if (v.length > LIMITS.evidenceMax) return bad('At most ' + LIMITS.evidenceMax + ' evidence links are allowed.');
  const out = [];
  for (const u of v) {
    const s = typeof u === 'string' ? u.trim() : '';
    if (!/^https?:\/\/\S+$/i.test(s) || s.length > LIMITS.urlMax) return bad('Each evidence link must be a http(s) URL under ' + LIMITS.urlMax + ' characters.');
    out.push(s);
  }
  return ok([...new Set(out)]);
}

export function validateFile(body, filerRole) {
  if (!body || typeof body !== 'object') return bad('Send a JSON body.');
  if (!REASONS[filerRole]?.includes(body.reason_code))
    return bad('reason_code must be one of: ' + (REASONS[filerRole] ?? []).join(', ') + '.');
  const d = cleanText(body.description, 'description');
  if (!d.ok) return d;
  const e = cleanEvidence(body.evidence_urls);
  if (!e.ok) return e;
  return ok({ reason_code: body.reason_code, description: d.value, evidence_urls: e.value });
}

export function validateRespond(body) {
  if (!body || typeof body !== 'object') return bad('Send a JSON body.');
  const t = cleanText(body.response_text, 'response_text');
  if (!t.ok) return t;
  const e = cleanEvidence(body.evidence_urls);
  if (!e.ok) return e;
  return ok({ response_text: t.value, response_evidence_urls: e.value });
}

export function validateDecision(body) {
  if (!body || typeof body !== 'object') return bad('Send a JSON body.');
  if (body.outcome !== 'upheld' && body.outcome !== 'rejected') return bad('outcome must be "upheld" or "rejected".');
  const n = cleanText(body.decision_note, 'decision_note');
  if (!n.ok) return n;
  return ok({ outcome: body.outcome, decision_note: n.value, proof_confirmed: body.proof_confirmed === true });
}

export function createEventsDb(sb) {

      return must(await sb.from('users').select('id,name,roles').eq('id', id).maybeSingle(), 'load user');

      if (error || !data?.user) return null;
      return must(await sb.from('users').select('id,name,roles').eq('auth_id', data.user.id).maybeSingle(), 'load user by token');
    },
    async getProduct(id) {
      return must(await sb.from('products').select('id,shop_id,status').eq('id', id).maybeSingle(), 'load product');

      return must(await sb.from('shops').select('id,owner_id,verification_status').eq('id', id).maybeSingle(), 'load shop');

        .eq('user_id', userId).eq('product_id', productId).eq('type', type).gte('created_at', sinceIso);
      if (error) throw new Error('check recent event: ' + error.message);
      return (count ?? 0) > 0;
    },
    async insertEvent(row) {
      must(await sb.from('event_logs').insert(row), 'save event');

// POST /api/events  (view and add_to_cart only; sign-in optional)
// - "purchase" events are NOT accepted here. The payment endpoint writes them, so nobody can fake sales.
// - shop_id is always taken from the product, never from the request.
// - A shop owner looking at their own product is not counted, and a signed-in user repeating the same
//   event within 30 seconds is counted once. Both keep the trending and performance numbers honest.
import { run, json, http, readJson, isPlainObject, UUID } from '../api/http.mjs';
import { currentUser } from '../auth/current-user.mjs';

export const CLIENT_EVENT_TYPES = ['view', 'add_to_cart'];
export const DEDUPE_SECONDS = 30;

export function handleRecordEvent(request, _params, dbOrFactory, { now = new Date() } = {}) {
  return run(dbOrFactory, async (db) => {
    const body = await readJson(request);
    if (!isPlainObject(body)) throw http(400, 'bad_json', 'Send a JSON object as the request body.');
    const type = typeof body.type === 'string' ? body.type.trim().toLowerCase() : '';
    if (!CLIENT_EVENT_TYPES.includes(type)) throw http(400, 'bad_type', 'type must be "view" or "add_to_cart".');
    if (typeof body.product_id !== 'string' || !UUID.test(body.product_id)) throw http(400, 'bad_product_id', 'product_id is not valid.');
    const productId = body.product_id.toLowerCase();

    // A bad token must not turn into an anonymous event, so identity is resolved before anything is written.
    const hasAuth = request.headers.get('authorization');
    const user = await currentUser(request, db);
    if (hasAuth && !user) throw http(401, 'not_signed_in', 'Please sign in again.');

    const product = await db.getProduct(productId);
    const shop = product ? await db.getShop(product.shop_id) : null;
    if (!product || !shop || product.status !== 'active' || shop.verification_status === 'rejected')
      throw http(404, 'product_not_found', 'No product with that id.');

    if (user && user.id === shop.owner_id) return json(200, { data: { recorded: false, reason: 'own_product' } });
    if (user) {
      const since = new Date(now.getTime() - DEDUPE_SECONDS * 1000).toISOString();
      if (await db.hasRecentEvent(user.id, productId, type, since)) return json(200, { data: { recorded: false, reason: 'duplicate' } });
    }
    await db.insertEvent({ user_id: user?.id ?? null, product_id: productId, shop_id: product.shop_id, type });
    return json(201, { data: { recorded: true } });
  }, 'events');
}

// Database access for the ledger. It reuses the orders database (users, shops) and adds one read of paid orders.
import { createOrdersDb } from '../orders/db.mjs';
import { LEDGER_DB_STATUSES, MAX_ROWS } from './rules.mjs';

const COLS = 'id,customer_id,shop_id,item_total,delivery_charge,commission_rate,commission_amount,payment_method,payment_status,status,placed_at,paid_at,delivered_at,released_at,refunded_at';
const CHUNK = 1000;   // Supabase returns at most 1000 rows per request

export function createLedgerDb(sb) {
  return {
    ...createOrdersDb(sb),
    // Reads every matching paid order in chunks. Returns up to MAX_ROWS + 1 rows so the caller can tell "too many".
    async listLedgerOrders({ shopId, status, startIso, endIso }) {
      const rows = [];
      for (let from = 0; rows.length <= MAX_ROWS; from += CHUNK) {
        let q = sb.from('orders').select(COLS).in('payment_status', status ? (status === 'held' ? ['held', 'paid'] : [status]) : LEDGER_DB_STATUSES);
        if (shopId) q = q.eq('shop_id', shopId);
        if (startIso) q = q.gte('placed_at', startIso);
        if (endIso) q = q.lt('placed_at', endIso);
        const { data, error } = await q.order('placed_at', { ascending: false }).order('id').range(from, from + CHUNK - 1);
        if (error) throw new Error('list ledger: ' + error.message);
        rows.push(...(data ?? []));
        if (!data || data.length < CHUNK) break;
      }
      return rows.slice(0, MAX_ROWS + 1);
    },
  };
}

import { parseLedgerQuery, dateRangeIso, toEntry, summarize, toCsv, dhakaToday, tooMany, MAX_ROWS } from './rules.mjs';

async function build(db, q, role, shopId) {
  const { start, end } = dateRangeIso(q);
  const rows = await db.listLedgerOrders({ shopId, status: q.status, startIso: start, endIso: end });
  if (rows.length > MAX_ROWS) throw tooMany();
  const shops = new Map((await db.getShopsByIds([...new Set(rows.map((r) => r.shop_id))])).map((s) => [s.id, s]));
  const entries = rows.map((r) => toEntry(r, { shop: shops.get(r.shop_id) }, role));
  return { entries, summary: summarize(entries) };
}

    } });
  }
  const from = (q.page - 1) * q.limit;
  return json(200, { data: { entries: entries.slice(from, from + q.limit), summary }, meta: { page: q.page, limit: q.limit, total: entries.length } });
}

// Pure ledger rules (PRD PAY-5): turn paid orders into ledger entries, add them up, write CSV. No database here.
import { http, UUID } from '../api/http.mjs';

export const LEDGER_STATUSES = ['held', 'released', 'refunded'];
// An order has a ledger entry once money has moved. `pending` and `failed` payments never reach the ledger.
export const LEDGER_DB_STATUSES = ['paid', 'held', 'released', 'refunded'];
export const MAX_ROWS = 10000;
export const PAGE_MAX = 200;
const DHAKA_OFFSET_HOURS = 6;

// Money seen by the platform for one order. Seller payout = item price minus commission (PAY-3).
export function toEntry(o, { shop, customer } = {}, role = 'seller') {
  const itemTotal = o.item_total;
  const commission = o.commission_amount ?? 0;
  const payout = itemTotal - commission;
  const paidOnline = o.payment_method === 'cod' ? o.delivery_charge : itemTotal + o.delivery_charge;
  const status = o.payment_status === 'paid' ? 'held' : o.payment_status;
  const entry = {
    order_id: o.id,
    shop_id: o.shop_id,
    shop_name: shop?.name ?? null,
    payment_method: o.payment_method,
    item_total: itemTotal,
    delivery_charge: o.delivery_charge,
    order_total: itemTotal + o.delivery_charge,
    paid_online: paidOnline,
    commission_rate: Number(o.commission_rate ?? 0),
    commission_amount: commission,
    seller_payout: payout,
    escrow_status: status,
    order_status: o.status,
    // What has actually settled: nothing is booked or paid out until the escrow is released.
    commission_booked: status === 'released' ? commission : 0,
    paid_to_seller: status === 'released' ? payout : 0,
    refunded_amount: status === 'refunded' ? paidOnline : 0,
    placed_at: o.placed_at ?? null,
    paid_at: o.paid_at ?? null,
    delivered_at: o.delivered_at ?? null,
    released_at: o.released_at ?? null,
    refunded_at: o.refunded_at ?? null,
  };
  if (role === 'admin') entry.customer_id = o.customer_id;
  return entry;
}

const sum = (rows, k) => rows.reduce((s, r) => s + (r[k] ?? 0), 0);
export function summarize(entries) {
  const by = (s) => entries.filter((e) => e.escrow_status === s);
  return {
    orders: entries.length,
    item_total: sum(entries, 'item_total'),
    in_escrow: sum(by('held'), 'seller_payout'),
    commission_booked: sum(entries, 'commission_booked'),
    paid_to_sellers: sum(entries, 'paid_to_seller'),
    refunded: sum(entries, 'refunded_amount'),
    orders_held: by('held').length,
    orders_released: by('released').length,
    orders_refunded: by('refunded').length,
  };
}

export function parseLedgerQuery(sp, { isAdmin }) {
  const errors = [];
  const format = (sp.get('format') ?? 'json').toLowerCase();
  if (!['json', 'csv'].includes(format)) errors.push('format must be json or csv.');
  const status = sp.get('status') ? sp.get('status').toLowerCase() : undefined;
  if (status && !LEDGER_STATUSES.includes(status)) errors.push(`status must be one of: ${LEDGER_STATUSES.join(', ')}.`);
  const from = sp.get('from') || undefined;
  const to = sp.get('to') || undefined;
  if (from && !isDate(from)) errors.push('from must be a date like 2026-11-01.');
  if (to && !isDate(to)) errors.push('to must be a date like 2026-11-30.');
  if (from && to && isDate(from) && isDate(to) && from > to) errors.push('from must not be after to.');
  const shopId = sp.get('shop_id') || undefined;
  if (shopId && isAdmin && !UUID.test(shopId)) errors.push('shop_id is not valid.');
  const int = (k, def, min, max) => { const s = sp.get(k); if (s == null || s === '') return def; if (!/^\d+$/.test(s) || Number(s) < min || Number(s) > max) { errors.push(`${k} must be between ${min} and ${max}.`); return def; } return Number(s); };
  const value = {
    format, status, from, to,
    shopId: isAdmin && shopId ? shopId.toLowerCase() : undefined,   // a seller can never pick another shop
    page: int('page', 1, 1, 100000), limit: int('limit', 50, 1, PAGE_MAX),
  };
  return { errors, value };
}

// Dates are in Dhaka time (UTC+6, no daylight saving). "to" is the whole last day.
export function dateRangeIso({ from, to }) {
  const start = from ? new Date(`${from}T00:00:00+06:00`).toISOString() : undefined;
  const end = to ? new Date(new Date(`${to}T00:00:00+06:00`).getTime() + 86400000).toISOString() : undefined;
  return { start, end };   // start inclusive, end exclusive
}

export function toDhaka(iso) {
  if (!iso) return '';
  const t = Date.parse(iso);
  if (Number.isNaN(t)) return '';
  return new Date(t + DHAKA_OFFSET_HOURS * 3600000).toISOString().slice(0, 16).replace('T', ' ');
}

// CSV cells that start with = + - @ are read as formulas by Excel. Shop names come from users, so neutralise them.
export function csvCell(v) {
  if (v === null || v === undefined) return '';
  let s = String(v);
  if (typeof v === 'string' && /^[=+\-@\t\r]/.test(s)) s = "'" + s;
  return /[",\r\n]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
}

}

// Without RESEND_API_KEY it only logs the message (fine for the demo). Set RESEND_API_KEY and MAIL_FROM to really send.
export async function sendMail({ to, subject, text }, { fetchImpl = fetch } = {}) {
  if (!to) return { sent: false, reason: 'no_recipient' };
  const key = process.env.RESEND_API_KEY;
  if (!key) {
    console.log('[mail stub] to=%s subject=%s', to, subject);
    return { sent: false, reason: 'stub' };
  }
  try {
    const res = await fetchImpl('https://api.resend.com/emails', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: 'Bearer ' + key },
      body: JSON.stringify({ from: process.env.MAIL_FROM || 'Dokan AI <onboarding@resend.dev>', to: [to], subject, text }),
      signal: AbortSignal.timeout(8000),
    });
    if (!res.ok) { console.error('mail failed: HTTP ' + res.status); return { sent: false, reason: 'http_' + res.status }; }
    return { sent: true };
  } catch (e) {
    console.error('mail failed:', e?.message ?? e);
    return { sent: false, reason: 'error' };
  }
}

const effectivePrice = (p) => (p.discount_price != null && p.discount_price < p.price ? p.discount_price : p.price);

function idParam(params) {
  const id = params?.id;
  if (typeof id !== 'string' || !UUID.test(id)) throw http(400, 'bad_id', 'The id in the URL is not valid.');
  return id.toLowerCase();

async function bodyOf(request) {
  const body = await readJson(request);
  if (!isPlainObject(body)) throw http(400, 'bad_json', 'Send a JSON object as the request body.');
  return body;
}
const RPC_ERRORS = {
  product_unavailable: [409, 'One of the products is no longer available.'],
  variant_unavailable: [409, 'That size or colour is no longer available.'],
  out_of_stock: [409, 'Not enough stock for one of the items.'],
  order_not_found: [404, 'No order with that id.'],
  already_cancelled: [409, 'This order is already cancelled.'],
  cannot_cancel: [409, 'This order can no longer be cancelled.'],
};
async function rpc(fn) {
  try { return await fn(); } catch (e) {
    const known = e.code === 'rpc' ? RPC_ERRORS[e.message] : null;
    if (known) throw http(known[0], e.message, known[1]);
    throw e;
  }
}

// The same order looks different to each party: only seller and admin see commission, only the customer's own
// side sees what is due online, and the address goes to the people who need it to deliver.
function orderView(o, role, { shop, customer } = {}) {
  const base = {
    id: o.id, shop_id: o.shop_id, shop_name: shop?.name ?? null, items: o.items ?? [],
    item_total: o.item_total, delivery_charge: o.delivery_charge, total: o.item_total + o.delivery_charge,
    amount_due_online: o.payment_method === 'cod' ? o.delivery_charge : o.item_total + o.delivery_charge,
    payment_method: o.payment_method, payment_status: o.payment_status, status: o.status,
    placed_at: o.placed_at, delivered_at: o.delivered_at ?? null, delivery_address: o.delivery_address ?? null, my_role: role,
  };
  if (role === 'customer') return base;
  return { ...base, customer: { id: o.customer_id, name: customer?.name ?? null }, commission_rate: Number(o.commission_rate), commission_amount: o.commission_amount, seller_payout: o.item_total - o.commission_amount };
}

// Who is this person for this order? Strangers get a 404 so order ids cannot be probed.
async function partyFor(db, user, order) {
  const shop = await db.getShop(order.shop_id);
  if (hasRole(user, 'admin')) return { role: 'admin', shop };
  if (shop && user.id === shop.owner_id) return { role: 'seller', shop };
  if (user.id === order.customer_id) return { role: 'customer', shop };
  throw http(404, 'order_not_found', 'No order with that id.');
}
async function loadOrder(db, id) {
  const order = await db.getOrder(id);

}

// POST /api/orders   (customer; one shop per order, the frontend splits a mixed cart)
export function handleCreateOrder(request, _params, dbOrFactory) {
  return run(dbOrFactory, async (db) => {
    const user = await requireUser(request, db);
    if (hasRole(user, 'admin')) throw http(403, 'admin_cannot_buy', 'Admins cannot place orders.');
    const { errors, value } = validateOrderInput(await bodyOf(request));
    if (errors.length) throw http(400, 'invalid_order', errors.join(' '));

    const products = new Map((await db.getProductsByIds([...new Set(value.items.map((i) => i.product_id))])).map((p) => [p.id, p]));
    const shopIds = new Set();
    const lines = [];
    for (const it of value.items) {
      const p = products.get(it.product_id);
      if (!p || p.status !== 'active') throw http(404, 'product_not_found', 'One of the products does not exist or is no longer available.');
      shopIds.add(p.shop_id);
      const variants = Array.isArray(p.variants) ? p.variants : [];
      const match = variants.filter((v) => lc(v?.size) === lc(it.size) && (!it.color || lc(v?.color) === it.color));
      if (match.length === 0) throw http(409, 'variant_unavailable', `${p.name}: that size or colour does not exist.`);
      if (match.length > 1) throw http(400, 'color_required', `${p.name}: choose a colour.`);
      if (!(Number(match[0].stock) >= it.qty)) throw http(409, 'out_of_stock', `${p.name}: only ${Math.max(0, Number(match[0].stock) || 0)} left in that size.`);
      lines.push({ product_id: p.id, name: p.name, size: match[0].size, color: lc(match[0].color), qty: it.qty, unit_price: effectivePrice(p), image_url: (p.image_urls ?? [])[0] ?? null });
    }
    if (shopIds.size > 1) throw http(400, 'multiple_shops', 'One order can only hold items from one shop. Place a separate order for each shop.');
    const shop = await db.getShop([...shopIds][0]);
    if (!shop || shop.verification_status === 'rejected') throw http(404, 'shop_not_found', 'This shop is not available.');
    if (shop.owner_id === user.id) throw http(400, 'own_shop', 'You cannot order from your own shop.');

    const itemTotal = lines.reduce((s, l) => s + l.unit_price * l.qty, 0);
    const order = await rpc(() => db.placeOrder({
      customerId: user.id, shopId: shop.id, items: lines, itemTotal, deliveryCharge: DELIVERY_CHARGE,
      method: value.payment_method, address: value.delivery_address,
    }));
    return json(201, { data: orderView(order, 'customer', { shop }) });
  }, 'orders');
}

// POST /api/orders/:id/pay   { gateway?: bkash|nagad|card, outcome?: success|failure|pending }   (the customer)

    const outcome = body.outcome === undefined ? 'success' : lc(body.outcome);

    if (!['pending', 'failed'].includes(order.payment_status)) throw http(409, 'already_paid', 'This order has already been paid.');

    const amount = order.payment_method === 'cod' ? order.delivery_charge : order.item_total + order.delivery_charge;
    const result = gateway.charge({ amount, gateway: gw, outcome });
    const shop = await db.getShop(order.shop_id);
    if (result.status === 'pending') return json(202, { data: { payment: { status: 'pending', reference: result.reference, amount, gateway: gw }, order: orderView(order, 'customer', { shop }) } });

    }

    // Success: the money is held in escrow and the commission is fixed on the order.
    const c = computeCommission(order.items);
    const paid = await db.markPaid(id, { payment_status: 'held', commission_rate: c.commission_rate, commission_amount: c.commission_amount, paid_at: now.toISOString() });
    if (!paid) throw http(409, 'already_paid', 'This order has already been paid.');
    // One purchase event per product feeds trending and performance. A logging problem must not undo the payment.
    try {
      await db.insertEvents([...new Set(order.items.map((i) => i.product_id))].map((pid) => ({ user_id: user.id, product_id: pid, shop_id: order.shop_id, type: 'purchase' })));
    } catch (e) { console.error('purchase events failed:', e?.message ?? e); }
    return json(200, { data: { payment: { status: 'success', reference: result.reference, amount, gateway: gw }, order: orderView(paid, 'customer', { shop }) } });
  }, 'orders');
}

// Pure order rules: input checks, prices, commission, and the allowed status changes. No database here.
import { UUID } from '../api/http.mjs';

export const DELIVERY_CHARGE = 80;
export const ESCROW_HOURS = 48;
export const PAYMENT_METHODS = ['sandbox', 'cod'];
export const LIMITS = { lines: 20, qty: 10, size: 20, color: 20, text: 120, page_max: 50 };
// Admin-editable later. The rate depends on the unit price after discount (PRD section 6.1).
export const COMMISSION_BANDS = [{ below: 500, rate: 0.02 }, { below: 1000, rate: 0.03 }, { below: Infinity, rate: 0.05 }];

export const commissionRate = (unitPrice) => COMMISSION_BANDS.find((b) => unitPrice < b.below).rate;

// Commission on the item price only, never on delivery. The amount is authoritative;
// the stored rate is the effective rate (exact when all lines fall in one band).
export function computeCommission(items) {
  const itemTotal = items.reduce((s, i) => s + i.unit_price * i.qty, 0);
  const amount = items.reduce((s, i) => s + Math.round(i.unit_price * i.qty * commissionRate(i.unit_price)), 0);
  return { item_total: itemTotal, commission_amount: amount, commission_rate: itemTotal ? Math.round((amount / itemTotal) * 100) / 100 : 0 };
}

const lc = (s) => String(s).trim().toLowerCase();
const text = (v, max) => (typeof v === 'string' && v.trim().length >= 1 && v.trim().length <= max ? v.trim().replace(/\s+/g, ' ') : null);

export function validateOrderInput(body) {
  const errors = [];
  const value = {};
  if (!Array.isArray(body.items) || body.items.length < 1 || body.items.length > LIMITS.lines) {
    errors.push(`items must be a list of 1 to ${LIMITS.lines} lines.`);
  } else {
    const merged = new Map();
    for (const it of body.items) {
      const size = typeof it?.size === 'string' ? it.size.trim() : '';
      const color = it?.color == null || it.color === '' ? '' : typeof it.color === 'string' ? lc(it.color) : null;
      if (typeof it?.product_id !== 'string' || !UUID.test(it.product_id) || !size || size.length > LIMITS.size || color === null || color.length > LIMITS.color ||
          !Number.isInteger(it.qty) || it.qty < 1 || it.qty > LIMITS.qty) {
        errors.push(`Each item needs product_id, size, an optional color and qty from 1 to ${LIMITS.qty}.`);
        break;
      }
      const key = [it.product_id.toLowerCase(), lc(size), color].join('|');
      const prev = merged.get(key);
      if (prev) prev.qty += it.qty; else merged.set(key, { product_id: it.product_id.toLowerCase(), size, color, qty: it.qty });
    }
    if (!errors.length) {
      const lines = [...merged.values()];
      if (lines.some((l) => l.qty > LIMITS.qty)) errors.push(`Quantity per line is at most ${LIMITS.qty}.`); else value.items = lines;
    }
  }
  const method = body.payment_method === undefined ? 'sandbox' : typeof body.payment_method === 'string' ? lc(body.payment_method) : '';
  if (!PAYMENT_METHODS.includes(method)) errors.push(`payment_method must be one of: ${PAYMENT_METHODS.join(', ')}.`); else value.payment_method = method;

  const a = body.delivery_address;
  if (!a || typeof a !== 'object' || Array.isArray(a)) errors.push('delivery_address is required: { name, phone, line1, city }.');
  else {
    const phone = typeof a.phone === 'string' ? a.phone.replace(/[\s-]/g, '') : '';
    const addr = { name: text(a.name, LIMITS.text), phone, line1: text(a.line1, 200), city: text(a.city, 60), area: a.area == null || a.area === '' ? null : text(a.area, 60) };
    if (!addr.name || !addr.line1 || !addr.city || !/^\+?[0-9]{10,15}$/.test(phone) || (a.area != null && a.area !== '' && !addr.area)) errors.push('delivery_address needs a name, a phone number (10 to 15 digits), line1 and city.');
    else value.delivery_address = addr;
  }
  return { errors, value };
}

export function parseListQuery(sp) {
  const errors = [];
  const as = (sp.get('as') ?? 'customer').toLowerCase();
  if (!['customer', 'seller', 'admin'].includes(as)) errors.push('as must be customer, seller or admin.');
  const status = sp.get('status');
  if (status && !['placed', 'accepted', 'packed', 'shipped', 'delivered', 'cancelled'].includes(status.toLowerCase())) errors.push('status is not valid.');
  const int = (k, def, min, max) => { const s = sp.get(k); if (s == null || s === '') return def; if (!/^\d+$/.test(s) || Number(s) < min || Number(s) > max) { errors.push(`${k} must be between ${min} and ${max}.`); return def; } return Number(s); };
  return { errors, value: { as, status: status ? status.toLowerCase() : undefined, page: int('page', 1, 1, 100000), limit: int('limit', 20, 1, LIMITS.page_max) } };
}

const NEXT = { placed: ['accepted', 'cancelled'], accepted: ['packed', 'cancelled'], packed: ['shipped', 'cancelled'], shipped: ['delivered'] };
const NEEDS_PAYMENT = new Set(['accepted', 'packed', 'shipped']);
const PAID = new Set(['paid', 'held', 'released']);

// actor: 'seller' | 'customer' | 'admin'. Returns null when allowed, otherwise { status, code, message }.
export function checkTransition({ order, to, actor }) {
  if (!(NEXT[order.status] ?? []).includes(to)) return { status: 409, code: 'bad_transition', message: `An order that is ${order.status} cannot become ${to}.` };
  if (actor === 'customer') {
    const ok = (to === 'cancelled' && order.status === 'placed') || (to === 'delivered' && order.status === 'shipped');
    if (!ok) return { status: 403, code: 'customer_cannot', message: 'Customers can cancel an order before it is accepted, or confirm that a shipped order arrived.' };
  }
  if (NEEDS_PAYMENT.has(to) && !PAID.has(order.payment_status)) return { status: 409, code: 'payment_required', message: 'The order has not been paid yet.' };
  return null;
}

import { UUID } from '../api/http.mjs';

export const CATEGORIES = ['clothing', 'jewelry', 'shoes', 'accessories'];
export const GENDERS = ['female', 'male', 'unisex'];
export const AGE_GROUPS = ['child', 'adult', 'elderly'];

export const LIMITS = {
  name: 120, description: 2000, custom_category: 40, colors: 10, color_len: 20, tags: 10, tag_len: 30,
  images: 8, url_len: 500, price_max: 1000000, variants: 30, size_len: 20, stock_max: 100000, q: 100, page_max: 50,
};

const isInt = Number.isInteger;
const lc = (s) => String(s).trim().toLowerCase();
const WORDISH = /^[\p{L}\p{N}][\p{L}\p{N} -]*$/u;

function wordList(v, field, max, maxLen, errors) {
  if (!Array.isArray(v) || v.length > max) { errors.push(`${field} must be a list of at most ${max} items.`); return undefined; }
  const out = [];
  for (const item of v) {
    if (typeof item !== 'string') { errors.push(`${field} must contain only text.`); return undefined; }
    const x = lc(item);
    if (x.length < 2 || x.length > maxLen || !WORDISH.test(x)) { errors.push(`${field}: "${item.slice(0, 30)}" is not a valid value.`); return undefined; }
    if (!out.includes(x)) out.push(x);
  }
  return out;
}

function validUrl(s) {
  if (typeof s !== 'string' || s.length > LIMITS.url_len) return false;
  try { const u = new URL(s); return u.protocol === 'http:' || u.protocol === 'https:'; } catch { return false; }
}

// partial=false: all required fields must be present (create). partial=true: only validate what was sent (patch).
export function validateProductInput(body, { partial = false } = {}) {
  const errors = [];
  const value = {};
  const has = (k) => Object.prototype.hasOwnProperty.call(body, k);
  const need = (k) => { if (!partial && !has(k)) errors.push(`${k} is required.`); return has(k); };

  if (need('name')) {
    const n = typeof body.name === 'string' ? body.name.replace(/\s+/g, ' ').trim() : '';
    if (!n || n.length > LIMITS.name) errors.push(`name must be 1 to ${LIMITS.name} characters.`); else value.name = n;
  }
  if (has('description')) {
    if (body.description === null) value.description = null;
    else if (typeof body.description !== 'string' || body.description.length > LIMITS.description) errors.push(`description must be text up to ${LIMITS.description} characters.`);
    else value.description = body.description.trim();
  }
  for (const [k, list] of [['category', CATEGORIES], ['gender', GENDERS], ['age_group', AGE_GROUPS]]) {
    if (need(k)) { const v = typeof body[k] === 'string' ? lc(body[k]) : ''; if (!list.includes(v)) errors.push(`${k} must be one of: ${list.join(', ')}.`); else value[k] = v; }
  }
  if (has('status')) { const v = typeof body.status === 'string' ? lc(body.status) : ''; if (!STATUSES.includes(v)) errors.push(`status must be one of: ${STATUSES.join(', ')}.`); else value.status = v; }
  if (has('custom_category')) {
    if (body.custom_category === null || body.custom_category === '') value.custom_category = null;
    else {
      const v = typeof body.custom_category === 'string' ? lc(body.custom_category) : '';
      if (!v || v.length > LIMITS.custom_category || !WORDISH.test(v)) errors.push('custom_category must be a short word such as "sari" or "three-piece".'); else value.custom_category = v;
    }
  }
  if (has('colors')) { const v = wordList(body.colors, 'colors', LIMITS.colors, LIMITS.color_len, errors); if (v) value.colors = v; }
  if (has('occasion_tags')) { const v = wordList(body.occasion_tags, 'occasion_tags', LIMITS.tags, LIMITS.tag_len, errors); if (v) value.occasion_tags = v; }
  if (has('image_urls')) {
    if (!Array.isArray(body.image_urls) || body.image_urls.length > LIMITS.images || !body.image_urls.every(validUrl)) errors.push(`image_urls must be at most ${LIMITS.images} http(s) links.`);
    else value.image_urls = body.image_urls;
  }
  if (need('price')) { if (!isInt(body.price) || body.price < 1 || body.price > LIMITS.price_max) errors.push(`price must be a whole number of BDT from 1 to ${LIMITS.price_max}.`); else value.price = body.price; }
  if (has('discount_price')) {
    if (body.discount_price === null) value.discount_price = null;
    else if (!isInt(body.discount_price) || body.discount_price < 1 || body.discount_price > LIMITS.price_max) errors.push('discount_price must be a whole number of BDT, or null.');
    else value.discount_price = body.discount_price;
  }
  if (need('variants')) {
    const v = body.variants;
    if (!Array.isArray(v) || v.length < 1 || v.length > LIMITS.variants) errors.push(`variants must be a list of 1 to ${LIMITS.variants} items.`);
    else {
      const out = []; const seen = new Set(); let ok = true;
      for (const x of v) {
        const size = typeof x?.size === 'string' ? x.size.trim() : '';
        const color = x?.color == null ? '' : (typeof x.color === 'string' ? lc(x.color) : null);
        if (!size || size.length > LIMITS.size_len || color === null || color.length > LIMITS.color_len || !isInt(x.stock) || x.stock < 0 || x.stock > LIMITS.stock_max) { ok = false; break; }
        const key = size.toLowerCase() + '|' + color;
        if (seen.has(key)) { errors.push('variants has the same size and colour twice.'); ok = false; break; }
        seen.add(key); out.push({ size, color, stock: x.stock });
      }
      if (ok) value.variants = out; else if (!errors.some((e) => e.startsWith('variants'))) errors.push('Each variant needs a size, an optional colour and a stock number of 0 or more.');
    }
  }
  return { errors, value };
}

const intParam = (s, name, min, max, errors) => {
  if (s == null || s === '') return undefined;
  if (!/^\d+$/.test(s)) { errors.push(`${name} must be a whole number.`); return undefined; }
  const n = Number(s);
  if (n < min || n > max) { errors.push(`${name} must be between ${min} and ${max}.`); return undefined; }
  return n;
};

  }
  const color = get('color'); if (color) v.color = color.toLowerCase().slice(0, LIMITS.color_len);

  const shop = get('shop_id'); if (shop) { if (!UUID.test(shop)) errors.push('shop_id is not valid.'); else v.shop_id = shop.toLowerCase(); }
  const min = intParam(get('min_price'), 'min_price', 0, LIMITS.price_max, errors); if (min !== undefined) v.min_price = min;

  v.page = intParam(get('page'), 'page', 1, 100000, errors) ?? 1;
  v.limit = intParam(get('limit'), 'limit', 1, LIMITS.page_max, errors) ?? 20;
  return { errors, value: v };
}

// Pure catalog search: takes rows already loaded from the database. No database calls here.

export const effectivePrice = (p) => (p.discount_price != null && p.discount_price < p.price ? p.discount_price : p.price);
const lc = (s) => String(s ?? '').trim().toLowerCase();
export const inStock = (p) => Array.isArray(p.variants) && p.variants.some((v) => Number(v?.stock) > 0);
export const isFeatured = (shop, now = new Date()) =>
  shop?.plan === 'premium' && (!shop.plan_expires_at || new Date(shop.plan_expires_at) > now);

const STOP = new Set(['a', 'an', 'the', 'for', 'and', 'or', 'in', 'on', 'of', 'to', 'my', 'me', 'i', 'with', 'under', 'below', 'above', 'show', 'find', 'need', 'want', 'bdt', 'tk', 'taka']);
const ALIASES = { saree: 'sari', sarees: 'sari', punjabi: 'panjabi', threepiece: 'three-piece' };

export function queryTokens(q) {
  return lc(q).replace(/three[\s]piece/g, 'three-piece').split(/[^\p{L}\p{N}-]+/u)
    .filter((t) => t && !STOP.has(t) && !/^\d+[kK]?$/.test(t));
}
function tokenVariants(t) {
  const out = new Set([t]);
  if (ALIASES[t]) out.add(ALIASES[t]);
  if (t.length > 3 && t.endsWith('s') && !t.endsWith('ss')) out.add(t.slice(0, -1));
  return [...out];
}
function matchesQuery(p, tokens) {
  if (!tokens.length) return true;
  const hay = [p.name, p.custom_category, p.category, ...(p.colors ?? []), ...(p.occasion_tags ?? [])].map(lc).join(' ');
  return tokens.every((t) => tokenVariants(t).some((v) => hay.includes(v)));
}

export function toPublicProduct(p, shop, now = new Date(), { withDescription = false } = {}) {
  return {
    id: p.id, shop_id: p.shop_id, name: p.name,
    ...(withDescription ? { description: p.description ?? null } : {}),
    category: p.category, custom_category: p.custom_category ?? null, gender: p.gender, age_group: p.age_group,
    colors: p.colors ?? [], occasion_tags: p.occasion_tags ?? [], image_urls: p.image_urls ?? [],
    price: p.price, discount_price: p.discount_price ?? null, price_now: effectivePrice(p),
    variants: p.variants ?? [], in_stock: inStock(p), status: p.status, created_at: p.created_at,
    shop: shop ? { id: shop.id, name: shop.name, city: shop.city ?? null, verified: shop.verification_status === 'verified', featured: isFeatured(shop, now) } : null,
  };
}

// Filters, sorts (premium first, then newest) and pages the catalog. A gender search also returns unisex items.
export function searchCatalog(rows, shops, params, now = new Date()) {
  const tokens = queryTokens(params.q);
  const hits = [];
  for (const p of rows) {
    const shop = shops.get(p.shop_id);
    if (!shop || shop.verification_status === 'rejected' || p.status !== 'active') continue;
    if (params.shop_id && p.shop_id !== params.shop_id) continue;
    if (params.category && p.category !== params.category) continue;
    if (params.gender && p.gender !== params.gender && p.gender !== 'unisex') continue;
    if (params.age_group && p.age_group !== params.age_group) continue;
    if (params.color && !(p.colors ?? []).some((c) => lc(c) === params.color)) continue;
    if (params.occasion && !(p.occasion_tags ?? []).some((o) => lc(o) === params.occasion)) continue;
    if (params.size && !(Array.isArray(p.variants) && p.variants.some((v) => lc(v?.size) === params.size && Number(v?.stock) > 0))) continue;
    const price = effectivePrice(p);
    if (params.min_price !== undefined && price < params.min_price) continue;
    if (params.max_price !== undefined && price > params.max_price) continue;
    if (params.in_stock && !inStock(p)) continue;
    if (!matchesQuery(p, tokens)) continue;
    hits.push({ p, shop, featured: isFeatured(shop, now) });
  }
  hits.sort((a, b) =>
    Number(b.featured) - Number(a.featured) ||
    new Date(b.p.created_at ?? 0) - new Date(a.p.created_at ?? 0) ||
    String(a.p.name).localeCompare(String(b.p.name)));
  const start = (params.page - 1) * params.limit;
  return { total: hits.length, items: hits.slice(start, start + params.limit).map((h) => toPublicProduct(h.p, h.shop, now)) };
}

const KNOWN_CATEGORIES = ['sari', 'kurti', 'three-piece', 'lehenga', 'panjabi', 'frock', 'shirt', 'formal shirt', 'pants', 'skirt', 'dress', 'salwar', 'burqa', 'hijab', 'fatua', 'shoes', 'heels', 'flats', 'sandals', 'oxford', 'earrings', 'necklace', 'bangles', 'ring', 'watch', 'clutch', 'bag', 'belt'];

const KNOWN_OCCASIONS = ['wedding', 'eid', 'party', 'formal', 'casual', 'festive', 'interview', 'holud'];
const esc = (s) => s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

function normalizeText(q) {
  let t = lc(q).replace(/three[\s]piece/g, 'three-piece');
  for (const [from, to] of Object.entries(ALIASES)) t = t.replace(new RegExp('(?<![\\p{L}\\p{N}-])' + from + '(?![\\p{L}\\p{N}-])', 'gu'), to);
  return t;
}
function findTerm(text, terms) {
  for (const term of [...terms].sort((a, b) => b.length - a.length)) {
    if (new RegExp('(?<![\\p{L}\\p{N}-])' + esc(term) + 's?(?![\\p{L}\\p{N}-])', 'u').test(text)) return term;
  }
  return null;
}

export function deriveLogFilters(params, rows) {
  const filters = {};
  const text = params.q ? normalizeText(params.q) : '';
  if (text) {
    const cats = new Set([...KNOWN_CATEGORIES, ...rows.map((r) => lc(r.custom_category)).filter(Boolean)]);
    const cat = findTerm(text, cats); if (cat) filters.custom_category = cat;
    const occ = new Set([...KNOWN_OCCASIONS, ...rows.flatMap((r) => (r.occasion_tags ?? []).map(lc))]);
    const o = findTerm(text, occ); if (o) filters.occasion = o;
  }
  const color = params.color ?? (text ? findTerm(text, new Set([...KNOWN_COLORS, ...rows.flatMap((r) => (r.colors ?? []).map(lc))])) : null);
  if (color) filters.color = color;
  if (params.occasion) filters.occasion = params.occasion;
  for (const k of ['category', 'gender', 'age_group', 'size', 'min_price', 'max_price']) if (params[k] !== undefined) filters[k] = params[k];
  return filters;
}

// Pure rules for shops: validation, Premium prices and dates. No database here.
export const LIMITS = { name_min: 2, name_max: 80, description: 1000, city: 60, policy: 1000, url: 300, reason_min: 3, reason_max: 500 };
// From the PRD (SF-12): 1, 3 or 6 months.
export const PREMIUM_PLANS = { 1: 1000, 3: 2500, 6: 5000 };
const NID = /^(\d{10}|\d{13}|\d{17})$/;   // Bangladesh NIDs have 10, 13 or 17 digits

function isHttp(s) {
  if (typeof s !== 'string' || s.length > LIMITS.url) return false;
  try { const u = new URL(s); return u.protocol === 'http:' || u.protocol === 'https:'; } catch { return false; }
}

// partial=false for create (name required); partial=true for edits.
export function validateShopInput(body, { partial = false } = {}) {
  const errors = [], value = {};
  const has = (k) => Object.prototype.hasOwnProperty.call(body, k);
  if (!partial && !has('name')) errors.push('name is required.');
  if (has('name')) {
    const n = typeof body.name === 'string' ? body.name.replace(/\s+/g, ' ').trim() : '';
    if (n.length < LIMITS.name_min || n.length > LIMITS.name_max) errors.push(`name must be ${LIMITS.name_min} to ${LIMITS.name_max} characters.`); else value.name = n;
  }
  for (const [k, max] of [['description', LIMITS.description], ['city', LIMITS.city], ['return_policy', LIMITS.policy]]) {
    if (!has(k)) continue;
    if (body[k] === null || body[k] === '') value[k] = null;
    else if (typeof body[k] !== 'string' || body[k].trim().length > max) errors.push(`${k} must be text up to ${max} characters.`);
    else value[k] = body[k].trim();
  }
  if (has('social_url')) {
    if (body.social_url === null || body.social_url === '') value.social_url = null;
    else if (!isHttp(body.social_url)) errors.push('social_url must be an http(s) link.'); else value.social_url = body.social_url;
  }
  if (has('nid_number')) {
    const n = typeof body.nid_number === 'string' || typeof body.nid_number === 'number' ? String(body.nid_number).replace(/[\s-]/g, '') : '';
    if (!NID.test(n)) errors.push('nid_number must have 10, 13 or 17 digits.'); else value.nid_number = n;
  }
  return { errors, value };
}

export function validateVerification(body) {
  const status = typeof body?.status === 'string' ? body.status.toLowerCase() : '';
  if (status !== 'verified' && status !== 'rejected') return { error: 'status must be "verified" or "rejected".' };
  if (status === 'verified') return { status };
  const reason = typeof body.reason === 'string' ? body.reason.trim() : '';
  if (reason.length < LIMITS.reason_min || reason.length > LIMITS.reason_max) return { error: `A rejection needs a reason of ${LIMITS.reason_min} to ${LIMITS.reason_max} characters.` };
  return { status, reason };
}

export function addMonths(date, months) {
  const d = new Date(date.getTime());
  const day = d.getUTCDate();
  d.setUTCDate(1);
  d.setUTCMonth(d.getUTCMonth() + months);
  const last = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth() + 1, 0)).getUTCDate();
  d.setUTCDate(Math.min(day, last));
  return d;
}

// Buying while Premium is still active extends from the current expiry; otherwise it starts now.
export function premiumExpiry(shop, months, now = new Date()) {
  const current = shop.plan === 'premium' && shop.plan_expires_at ? new Date(shop.plan_expires_at) : null;
  return addMonths(current && current > now ? current : now, months);
}

export function shapeCatalog(shopRows, productRows, badgedOwnerIds) {
  const shops = shopRows
    .filter((s) => s.verification_status === 'verified')
    .map((s, i) => ({
      id: i + 1, uuid: s.id, name: s.name, city: s.city || 'Bangladesh',
      rating: 4.5, followers: 0,                       // no ratings/followers in the database yet
      verified: true, fraud: badgedOwnerIds.has(s.owner_id), premium: s.plan === 'premium',
      policy: s.return_policy || 'Ask the seller about returns and exchanges.',
    }));
  const shopNo = new Map(shops.map((s) => [s.uuid, s.id]));

  const products = productRows
    .filter((p) => p.status === 'active' && shopNo.has(p.shop_id))
    .map((p) => {
      const variants = Array.isArray(p.variants) ? p.variants : [];
      const stock = variants.reduce((t, v) => t + (v.stock > 0 ? v.stock : 0), 0);
      const sizes = [...new Set(variants.filter((v) => v.stock > 0).map((v) => v.size))];
      const price = p.discount_price ?? p.price;
      return {
        p, stock, sizes, price,
      };
    })
    .filter((x) => x.stock > 0)                         // out-of-stock products are not listed
    .map(({ p, stock, sizes, price }, i) => ({
      id: i + 1, uuid: p.id, shopUuid: p.shop_id,
      name: p.name,
      cat: cap(p.custom_category || p.category),
      type: p.category,
      gender: p.age_group === 'child' ? 'Children' : p.age_group === 'elderly' ? 'Elderly' : p.gender === 'male' ? 'Men' : 'Women',
      unisex: p.gender === 'unisex',
      color: (p.colors && p.colors[0]) || '',
      colors: p.colors || [],
      price, oldPrice: p.discount_price != null && p.discount_price < p.price ? p.price : null,
      shop: shopNo.get(p.shop_id),
      occasion: cap((p.occasion_tags && p.occasion_tags[0]) || ''),
      tags: p.occasion_tags || [],
      sizes, stock,
      image: (p.image_urls && p.image_urls[0]) || null,
      img: (p.image_urls && p.image_urls[0]) || null,   // ProductArt uses p.img for the real photo
      desc: p.description || `${p.name} from a verified small seller. Delivery by courier across Bangladesh.`,
    }));
  return { shops, products };
}

let loading = null;
export function loadCatalog() {
  if (!URL || !KEY) return Promise.resolve(false);
  if (loading) return loading;
  loading = (async () => {
    try {
      const [shopRows, productRows, badges] = await Promise.all([
        get('shops?select=id,owner_id,name,city,verification_status,plan,return_policy'),
        get('products?select=*&status=eq.active&order=created_at.asc,id.asc&limit=1000'),
        get('badges?select=user_id&status=eq.active'),
      ]);
      const { shops, products } = shapeCatalog(shopRows, productRows, new Set(badges.map((b) => b.user_id)));
      if (!products.length) return false;
      SHOPS.splice(0, SHOPS.length, ...shops);
      PRODUCTS.splice(0, PRODUCTS.length, ...products);
      return true;
    } catch (e) {
      console.error('Could not load catalog, using built-in sample data:', e);
      loading = null;
      return false;
    }
  })();
  return loading;
}

/** Budget guardrails from the PRD (section 5.3). Swap this for a backend call later. */
export function buildOutfit(key, budget) {
  const items = CATALOG[key].items;
  const lowest = items.reduce((s, i) => s + i.o[0].price, 0);
  const highest = items.reduce((s, i) => s + i.o[2].price, 0);
  const mid = Math.round((lowest + highest) / 2);
  if (!(budget > 0)) return { error: `Please enter a budget in BDT, or choose low / medium / high: ${bdt(lowest)} · ${bdt(mid)} · ${bdt(highest)}.` };
  if (budget < items[items.length - 1].o[0].price)
    return { error: `Honestly, a complete outfit for ${bdt(budget)} isn't achievable. The lowest realistic full look is about ${bdt(lowest)}. I can show a single accessory within budget, or you can raise the budget.` };
  const pick = items.map(() => 0);
  for (let i = pick.length - 1; i >= 0 && sum(key, pick) > budget; i--) pick[i] = -1;
  for (let up = true; up; ) {
    up = false;
    pick.forEach((t, i) => {
      if (t < 0 || t > 1 || budget >= highest * 1.25) return;
      const d = items[i].o[t + 1].price - items[i].o[t].price;
      if (sum(key, pick) + d <= budget) { pick[i]++; up = true; }
    });
  }
  const note = budget > highest * 1.8 ? `Dokan AI focuses on affordable small-business fashion, so I won't pad the look just to reach ${bdt(budget)}. Here are the best premium-range options we actually have.` : '';
  return { pick, note };
}

export function swapItem(key, pick, i, budget) {
  const n = CATALOG[key].items[i].o.length;
  for (let k = 1; k <= n; k++) {
    const next = pick.slice(); next[i] = ((pick[i] < 0 ? 0 : pick[i]) + k) % n;
    if (sum(key, next) <= budget) return next;
  }
  return pick;
}

  let res;
  try {
    res = await fetch(`${BASE}/functions/v1/stylist-chat`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ message, conversation_id: conversationId || undefined }),
    });
  } catch {
    throw new Error('Could not reach the stylist. Check your internet connection.');
  }
  let json = null;
  try { json = await res.json(); } catch { /* not JSON */ }
  if (!res.ok) throw new Error(json?.error?.message || `The stylist returned an error (${res.status}).`);
  return json.data;
}

# Complaints API (for the frontend)

All responses are `{ "data": ... }` or `{ "error": { "code", "message" } }`.

**Who is calling:** send `Authorization: Bearer <session.access_token>`. In demo mode only, the header `x-user-id: <user uuid>` still works (it is ignored once a Bearer header is present).
A bad or expired Bearer token returns 401 and never falls back to the demo header. Set `DISABLE_DEV_USER_HEADER=true` on any deployed site to switch the demo header off.
Admin endpoints require a user whose `roles` contain `admin`.

## Customer or seller

| Call | Body | Notes |
|---|---|---|
| `POST /api/orders/:id/complaints` | `{ reason_code, description, evidence_urls? }` | Only the customer or the shop owner of that order. One per person per order. Max 5 per 24 hours. Returns 201. |
| `GET /api/complaints/mine` | none | Complaints I filed and complaints against me. Decision note appears only after the decision. |
| `POST /api/complaints/:id/respond` | `{ response_text, evidence_urls? }` | Only the accused. Only while status is `submitted` (one response). |

Reason codes by role:
- customer: `not_delivered`, `not_as_described`, `wrong_size`, `other`
- seller: `refused_delivery`, `false_claim`, `other`

`description` and `response_text` need 10 to 2000 characters. `evidence_urls` is a list of up to 10 `http(s)` links.

## Admin

| Call | Body | Notes |
|---|---|---|
| `GET /api/admin/complaints?status=open` | none | `open` = submitted + responded (oldest first). Or one of `submitted`, `responded`, `upheld`, `rejected`, `appealed`. Each row has `response_due_at` and `response_overdue`. |
| `GET /api/admin/complaints/:id` | none | Both sides, the order, the accused history (`upheld_before`, `has_active_badge`). |
| `PATCH /api/admin/complaints/:id/decision` | `{ outcome, decision_note, proof_confirmed? }` | `outcome` is `upheld` or `rejected` (there is no "partially upheld"). `decision_note` is required (10+ characters). |

A badge is issued only when the outcome is `upheld`, `proof_confirmed` is `true` **and** the server has `ISSUE_BADGES=true`.
Badges are switched off by default. The response says what happened in `badge_issued` and `badge_note`.

## Public badge status

`GET /api/users/:id/badge` and `GET /api/shops/:id/badge` return
`{ has_badge, label, level, roles }`. The label is "Verified complaints". Nothing about the complaint or the other party is exposed.

## Error codes

| HTTP | code | Meaning |
|---|---|---|
| 400 | `bad_id`, `invalid_complaint`, `invalid_response`, `invalid_decision`, `bad_status`, `self_order`, `note_required` | Bad input |
| 401 | `not_signed_in` | Missing or invalid login (bad token, or unknown `x-user-id` in demo mode) |
| 403 | `not_your_order`, `not_accused`, `admin_only`, `conflict_of_interest` | Not allowed |
| 404 | `order_not_found`, `complaint_not_found`, `shop_not_found` | Does not exist |
| 409 | `already_filed`, `order_cancelled`, `not_open_for_response`, `already_decided` | Wrong state |
| 429 | `too_many_complaints` | Rate limit |
| 500 | `internal` | Something went wrong (details are only in the server log) |

# Dokan AI: API contract

This is the agreement between Frontend, Backend and AI/Analytics. It follows PRD section 12 and records where the built API differs from it. If you need to change a field, tell the other two first and update this file.

Detailed docs: `products-api.md`, `shops-api.md`, `orders-api.md`, `ledger-api.md`, `complaints-api.md`, `me-api.md`.

## Conventions

- JSON over REST, all paths under `/api`. Success is `{ "data": ... }` (lists add `"meta": { page, limit, total }`). Failure is `{ "error": { "code", "message" } }`.
- IDs are UUID strings. Money is a whole number of BDT. Timestamps are ISO 8601 in UTC; the app shows them in Asia/Dhaka.
- Enums are exact lowercase strings (see the PRD, section 12.3). Order status: `placed, accepted, packed, shipped, delivered, cancelled`. Payment status: `pending, paid, held, released, refunded, failed`.
- List endpoints take `page` (default 1) and `limit` (default 20, max 50).

## Auth (a change from the PRD)

The PRD lists `POST /api/auth/signup` and `/api/auth/login`. They are NOT built. Sign up and log in go straight to Supabase Auth from the browser (`supabase.auth.signUp`, `signInWithPassword`). A database trigger (`scripts/security.sql`) creates the matching `users` row, and only the roles `customer` and `seller` can be chosen at signup. Admins are promoted by hand in SQL.

Every API call sends `Authorization: Bearer <session.access_token>`. `lib/auth/current-user.mjs` is the only place that decides who is calling. The demo header `x-user-id` works only while `DISABLE_DEV_USER_HEADER` is not `true`; set it to `true` on any deployed site.

A bad or expired token returns 401 `not_signed_in` and never falls back to the demo header.

## Endpoints

Who can call: P = public, U = any signed-in user, C = customer, S = seller (shop owner), A = admin.

| Method and path | Who | Purpose |
|---|---|---|
| GET `/api/me` | U | Current user, roles and the seller's shop |
| GET `/api/products` | P | Search and filter. Premium shops sort first. Each search is logged |
| GET `/api/products/:id` | P | One product |
| POST `/api/products` | S | Create a product |
| PATCH `/api/products/:id` | S, A (status only) | Edit, or hide with `status: "hidden"`. There is no delete |
| POST `/api/shops` | U | Create a shop (starts as pending, grants the seller role) |
| GET `/api/shops/mine` | U | My shop, or `null` |
| GET `/api/shops/:id` | P | Shop. Owner and admin see more fields |
| PATCH `/api/shops/:id` | S | Edit shop (a rejected shop goes back to pending) |
| POST `/api/shops/:id/premium` | S | Buy Premium for 1, 3 or 6 months (sandbox) |
| GET `/api/admin/shops` | A | Verification queue |
| PATCH `/api/admin/shops/:id/verification` | A | Verify, or reject with a reason |
| POST `/api/events` | P | Record `view` or `add_to_cart`. `purchase` is refused; the payment writes it |
| POST `/api/orders` | C | Place an order. One shop per order |
| GET `/api/orders?as=customer\|seller\|admin` | U | List orders |
| GET `/api/orders/:id` | C, S, A | One order. Others get 404 |
| PATCH `/api/orders/:id/status` | C, S, A | Move the order along, or cancel |
| POST `/api/orders/:id/pay` | C | Sandbox payment |
| POST `/api/admin/orders/:id/refund` | A | Full refund of money still held |
| POST `/api/admin/payouts/release` | A | Release escrow 48 hours after delivery |
| POST `/api/orders/:id/complaints` | C, S | File a complaint |
| GET `/api/complaints/mine` | U | My complaints |
| POST `/api/complaints/:id/respond` | accused | Respond once |
| GET `/api/admin/complaints`, GET `/api/admin/complaints/:id` | A | Complaint queue and detail |
| PATCH `/api/admin/complaints/:id/decision` | A | `upheld` or `rejected`, with a written note |
| GET `/api/users/:id/badge`, GET `/api/shops/:id/badge` | P | Badge label and level only |
| GET `/api/seller/analytics` | S, A | Seller dashboard. A seller always gets their own shop. An admin adds `?shop_id=` |
| GET `/api/seller/ledger` | S | Own shop's payment ledger, JSON or CSV (`?format=csv`) |
| GET `/api/admin/ledger` | A | All shops' ledger, JSON or CSV. Add `?shop_id=` to narrow |
| GET `/api/admin/insights` | A | Platform trends and demand versus supply |
| GET `/api/top-sellers` | P | Top shops |

## Not built or different from the PRD

- `POST /api/auth/signup` and `/login`: replaced by Supabase Auth (above).
- `POST /api/ai/stylist/chat` and `/cart`: the stylist runs as the Supabase function `stylist-chat`, which the frontend calls directly (`lib/stylistApi.js`).
- Complaint decisions accept `upheld` and `rejected` only. There is no "partially upheld".
- Fraud badges are issued only when the server has `ISSUE_BADGES=true`.
- Not built: partial refunds, scheduled 48-hour payout (an admin calls the release endpoint), the 12-hour seller accept timeout, expiry of unpaid orders, notifications, order chat, reviews, wishlist, image upload, admin two-factor login.

## Rules the code follows

- Prices and totals for an order come from the server. The browser never sends a price. Delivery is 80 BDT.
- Commission is on the unit price after discount, never on delivery: 2% under 500, 3% under 1,000, 5% from 1,000. It is stored on the order when the payment succeeds.
- The shop on an event or order comes from the product, never from the client.
- A seller cannot order from their own shop. Admins cannot place orders or own shops.
- Escrow is not released while a complaint is submitted, responded, appealed or upheld.
- Only an admin can create, change or remove a badge. Nothing is automatic.
- Stock changes and money changes run as database functions, so two buyers cannot take the last item. Only the server's secret key can call them.
- The sandbox payment outcome (`success`, `failure`, `pending`) is chosen by the caller so the team can demo failures. Remove that before any real launch, and keep the "sandbox" banner visible (PRD PAY-6).

# Payment ledger API (PRD PAY-5)

Send `Authorization: Bearer <session.access_token>`. Errors look like `{ "error": { "code", "message" } }`.
**Sandbox only: no real money moves. Show the sandbox banner (PAY-6).**

One ledger entry per order whose payment succeeded (`held`, `released` or `refunded`). Pending and failed payments are not in the ledger.

| Endpoint | Who | Notes |
|---|---|---|
| `GET /api/seller/ledger` | Seller | Their own shop only. `shop_id` is ignored. A seller with no shop gets an empty ledger. |
| `GET /api/admin/ledger` | Admin | Every shop. Adds `customer_id` to each entry. `?shop_id=<uuid>` narrows to one shop. |

**Query (both):** `format=json|csv` (default json), `status=held|released|refunded`, `from=YYYY-MM-DD`, `to=YYYY-MM-DD` (both inclusive, Dhaka days, filter on the order's placed date), `page` and `limit` (json only; limit 1 to 200, default 50).

**JSON reply:** `{ data: { entries: [...], summary }, meta: { page, limit, total } }`. The summary covers every matching row, not just the page.

**Entry fields:** `order_id, shop_id, shop_name, payment_method, item_total, delivery_charge, order_total, paid_online, commission_rate, commission_amount, seller_payout, escrow_status, order_status, commission_booked, paid_to_seller, refunded_amount, placed_at, paid_at, delivered_at, released_at, refunded_at` (+ `customer_id` for admin).

- `seller_payout` = `item_total` - `commission_amount` (PAY-3). It is what the seller gets once the money is released.
- `commission_booked` and `paid_to_seller` are 0 until the escrow is `released`. `refunded_amount` is the amount paid online, once `refunded`.
- `paid_online` is `order_total`, or only the delivery charge for cash on delivery.
- Orders paid before the ledger columns were added have `null` for `paid_at`, `released_at` and `refunded_at`.

**Summary fields:** `orders, item_total, in_escrow` (seller payout still held), `commission_booked, paid_to_sellers, refunded, orders_held, orders_released, orders_refunded`.

**CSV download:** `GET /api/seller/ledger?format=csv`. The reply is a file (`Content-Disposition: attachment`), UTF-8 with a BOM so Excel shows Bangla names. Dates are in Dhaka time. The frontend must fetch it with the Bearer header and save the blob; a plain link will not send the token:

```js
const res = await fetch('/api/seller/ledger?format=csv', { headers: { Authorization: `Bearer ${token}` } });
const url = URL.createObjectURL(await res.blob());
Object.assign(document.createElement('a'), { href: url, download: 'ledger.csv' }).click();
URL.revokeObjectURL(url);
```

**Limit:** more than 10,000 matching rows returns 413 `too_many_rows`; narrow with `from` and `to`.
Codes: `seller_only` (403), `admin_only` (403), `not_signed_in` (401), `bad_query` (400), `too_many_rows` (413).

# Orders and sandbox payment API

Send `Authorization: Bearer <session.access_token>` (demo mode only: `x-user-id`). Errors look like `{ "error": { "code", "message" } }`.
**Sandbox only: no real money moves. Show the sandbox banner (PAY-6).**

| Endpoint | Who | Notes |
|---|---|---|

One shop per order: the frontend splits a mixed cart. Prices come from the server (discount applied), delivery is 80 BDT. Returns 201 with status `placed`, payment `pending`. |
| `POST /api/orders/:id/pay` | The customer | `{ gateway?: "bkash"\|"nagad"\|"card", outcome?: "success"\|"failure"\|"pending" }`. 200 paid (money held in escrow), 402 failed (retry allowed), 202 pending. COD orders pay only the delivery charge online. |

| `POST /api/admin/orders/:id/refund` | Admin | Full refund of money still held. |
| `POST /api/admin/payouts/release` | Admin | Releases escrow for orders delivered 48 hours ago or more, skipping any with an open or upheld complaint. |

**Status rules:** placed → accepted → packed → shipped → delivered. Cancel is possible until shipped.
Seller and admin do every step; the customer may cancel a `placed` order, or confirm a `shipped` order arrived (this releases the money at once unless a complaint is open).
Accepting, packing and shipping need a paid order (`payment_status` held or released).

Codes to handle: `out_of_stock`, `variant_unavailable`, `color_required`, `multiple_shops`, `own_shop`, `payment_required`, `already_paid`, `customer_cannot`, `bad_transition`, `status_changed`.

# Products API

All replies are `{ "data": ... }` or `{ "error": { "code", "message" } }`.
Who is calling comes from `lib/auth/current-user.mjs` only (Bearer token, or the demo header `x-user-id`).

Full product with `description`. Hidden products return 404 unless you are the owner or an admin.

Send only the fields to change. A product cannot move to another shop.

# Shops API

Replies are `{ "data": ... }` or `{ "error": { "code", "message" } }`. Identity comes from `lib/auth/current-user.mjs`.
Run `scripts/shops.sql` once first (adds `shops.rejection_note`).

## POST /api/shops   (any signed-in user)
Body: `name` (required), `description, city, social_url, nid_number, return_policy`.
Creates the shop as **pending** and gives the user the `seller` role. One shop per user (409 `shop_exists`). Admins cannot own shops.
`nid_number` is 10, 13 or 17 digits. It is stored but never sent back (owners get `nid_set: true`).

## GET /api/shops/mine   (signed in)
The caller's shop, or `{ data: null }` if they have none yet. Use this to decide between "Become a seller" and the seller dashboard.

## GET /api/shops/:id   (public)
Public view: `id, name, description, city, social_url, return_policy, verified, featured, created_at`.
Owner and admin also get `verification_status, plan, plan_expires_at, rejection_note`. Admin also gets `owner_id, nid_number`.
Rejected shops are hidden from everyone except the owner and admins.

## PATCH /api/shops/:id   (owner)
Editable: `name, description, city, social_url, return_policy, nid_number`.
`plan`, `verification_status` and `owner_id` are refused. Editing a **rejected** shop sends it back to **pending** (resubmit).
A verified shop that changes its name or NID also goes back to pending.

## POST /api/shops/:id/premium   (owner, sandbox payment)
Body: `{ "months": 1 | 3 | 6 }` costs 1,000 / 2,500 / 5,000 BDT. Sets `plan: premium` and `plan_expires_at`.
Buying while Premium is active extends from the current expiry. Reply: `{ shop, paid, months, sandbox: true }`.
An expired plan shows as `free` automatically; no data is deleted.

## GET /api/admin/shops?status=pending|verified|rejected&page=&limit=   (admin)
The review queue, oldest first, including `nid_number` and `owner_id`. Reply has `meta: { page, limit, total }`.

## PATCH /api/admin/shops/:id/verification   (admin)
Body: `{ "status": "verified" }` or `{ "status": "rejected", "reason": "..." }` (reason 3 to 500 characters, required).
The reason is saved on the shop (the owner sees it as `rejection_note`) and emailed to the owner if an email exists.
Reply: `{ shop, emailed }`. Without `RESEND_API_KEY` the email is only logged, so `emailed` is false.

## Not built
Downgrade by hand, a payment ledger (no record of who paid), the free-plan product cap, logo/cover images, payout account, and image upload.
